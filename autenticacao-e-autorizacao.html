    8. Autenticação e autorização

    Duas questões distintas surgem a cada requisição:

    • Quem é o usuário? É a autenticação: o usuário se identificou com um nome de usuário e uma senha, e cada requisição seguinte precisa provar que vem dele;
    • O usuário tem permissão para acessar esta página? É a autorização: no estudo de caso, apenas um usuário com o papel ADMIN gerencia médicos e clientes.

    Como o servidor não mantém estado (Capítulo 6), a prova de identidade deve ser incluída em cada requisição. Usa-se um token JWT ([JSON Web Token]): após o login, o servidor gera um token contendo a identidade do usuário e o assina com uma chave secreta. O token é armazenado em um cookie; o navegador o reenvia a cada requisição; o servidor verifica a assinatura. O usuário pode ler seu token, mas não pode modificá-lo: qualquer modificação invalida a assinatura.

    [ASP.NET Core] distingue claramente as duas questões:

    • a autenticação é tratada por um middleware ([UseAuthentication]) e por um esquema de autenticação: neste caso, o esquema [JwtBearer], fornecido pelo pacote [NuGet] [Microsoft.AspNetCore.Authentication.JwtBearer] . A cada requisição, ele procura um token, o verifica e dele deriva o usuário, [HttpContext.User];
    • a autorização é tratada pelo atributo [[Authorize]], verificado pelo middleware [UseAuthorization]: ele compara as exigências da action com o que se sabe sobre o usuário.

    Os exemplos deste capítulo usam dois pacotes [NuGet]: [Microsoft.AspNetCore.Authentication.JwtBearer] e [BCrypt.Net-Next] (hash de senhas). Eles são declarados no arquivo de projeto; o comando [dotnet build] (ou [dotnet run]) os baixa de nuget.org durante a primeira compilação: é necessária, portanto, uma conexão com a Internet.

    namespace Example25.Auth;

public record User(string Login, string Name, string Role, string PasswordHash);

// os usuários, em memória (o próximo capítulo os colocará em um banco de dados)
public class UsersService
{
    private readonly List<User> users =
    [
        // uma senha NUNCA é guardada, apenas seu hash
        Create("marie", "Marie Pélissier", "USER", "marie"),
        Create("admin", "Administrator", "ADMIN", "admin"),
    ];

    public User? Find(string login) =>
        users.Find(u => u.Login == login);

    // bcrypt, com custo 10: 2^10 rodadas de cálculo, deliberadamente lento
    private static User Create(string login, string name, string role, string password) =>
        new(login, name, role, BCrypt.Net.BCrypt.HashPassword(password, 10));
}
    • linha 3: um usuário, com seu papel;
    • linhas 10–12: uma senha nunca é armazenada; apenas seu hash é mantido — uma impressão digital calculada por uma função unidirecional. É possível verificar se uma senha corresponde a uma impressão digital, mas não recuperar a senha a partir da impressão digital;
    • linhas 19–20: [BCrypt.Net.BCrypt.HashPassword(password, 10)] calcula o hash usando o algoritmo [bcrypt]. O 10 representa o “custo”: o [bcrypt] é intencionalmente lento (2¹⁰ ciclos de cálculo) para desencorajar quem tente milhões de senhas. Dois hashes da mesma senha são diferentes (um “salt” aleatório é adicionado): [BCrypt.Verify] é responsável por verificá-los.

    using System.Text;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Example25.Auth;

public class AuthService(UsersService users, IConfiguration configuration)
{
    // o nome do cookie que carrega o token
    public const string Cookie = "token";

    // a chave que assina os tokens, lida de appsettings.json (seção Jwt)
    public static SymmetricSecurityKey Key(IConfiguration configuration) =>
        new(Encoding.UTF8.GetBytes(configuration["Jwt:Secret"]!));

    // null se o login for desconhecido ou a senha estiver errada
    public User? Validate(string? login, string? password)
    {
        var user = users.Find(login ?? "");
        if (user is null || password is null)
        {
            return null;
        }
        return BCrypt.Net.BCrypt.Verify(password, user.PasswordHash) ? user : null;
    }

    // o token JWT do usuário, assinado com a chave secreta
    public string CreateToken(User user)
    {
        var descriptor = new SecurityTokenDescriptor
        {
            // o conteúdo do token: as "claims" (declarações sobre o usuário)
            Claims = new Dictionary<string, object>
            {
                ["sub"] = user.Login,
                ["name"] = user.Name,
                ["role"] = user.Role,
            },
            Expires = DateTime.UtcNow.AddMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
            SigningCredentials = new SigningCredentials(Key(configuration), SecurityAlgorithms.HmacSha256),
        };
        return new JsonWebTokenHandler().CreateToken(descriptor);
    }
}

    using System.Text.Encodings.Web;
using System.Text.Unicode;
using Example25.Auth;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.FileProviders;
using Microsoft.IdentityModel.Tokens;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddControllersWithViews();
builder.Services.AddSingleton(HtmlEncoder.Create(UnicodeRanges.All));
builder.Services.AddSingleton<UsersService>();
builder.Services.AddSingleton<AuthService>();

// autenticação: um token JWT, verificado a cada requisição
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.MapInboundClaims = false; // mantém os nomes das claims como estão ("sub", "role"...)
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = AuthService.Key(builder.Configuration), // a assinatura deve ser a nossa
            ValidateLifetime = true,                                   // o token não deve estar expirado
            ValidateIssuer = false,
            ValidateAudience = false,
            NameClaimType = "name", // User.Identity.Name
            RoleClaimType = "role", // [Authorize(Roles = ...)], User.IsInRole(...)
        };
        options.Events = new JwtBearerEvents
        {
            // 1. o token é lido do cookie (e não do cabeçalho Authorization)
            OnMessageReceived = context =>
            {
                context.Token = context.Request.Cookies[AuthService.Cookie];
                return Task.CompletedTask;
            },
            // 2. página protegida sem token válido: direto para a página de login
            OnChallenge = context =>
            {
                context.HandleResponse(); // construímos a resposta nós mesmos (sem 401)
                context.Response.Redirect("/login");
                return Task.CompletedTask;
            },
        };
    });
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseStaticFiles(new StaticFileOptions
{
    FileProvider = new PhysicalFileProvider(Path.GetFullPath(Path.Combine(builder.Environment.ContentRootPath, "..", "common", "bootstrap"))),
    RequestPath = "/bootstrap",
});
app.UseAuthentication(); // quem é? -> HttpContext.User
app.UseAuthorization();  // é permitido? -> [Authorize]

app.MapControllers();
app.Run();

    Vamos comentar este código:

    • linhas 16–17: declaramos o esquema de autenticação [JwtBearer] e o definimos como esquema padrão;
    • linha 19: por padrão, o [JwtBearer] renomeia certas claims usando nomes longos herdados de tecnologias da Microsoft ([role] se tornaria [http://schemas.microsoft.com/ws/2008/06/identity/claims/role]). Mantemos os nomes do token;
    • Linhas 20–26: as verificações feitas em cada token: a assinatura (com a nossa chave) e a data de expiração. Nossos tokens não especificam nem o emissor ([issuer]) nem o destinatário ([audience]), portanto não os verificamos;
    • linhas 27–28: qual claim contém o nome do usuário ([User.Identity.Name]) e qual contém seu papel ([[Authorize(Roles = ...)]], [User.IsInRole(...)]);
    • linhas 33–36: os eventos de [JwtBearer] permitem personalizar seu comportamento. Por padrão, ele procura o token no cabeçalho [Authorization: Bearer ...] da requisição: é assim que funciona um cliente [JavaScript] ou um aplicativo móvel. Um navegador não adiciona esse cabeçalho por conta própria: o token é recuperado do cookie;
    • linhas 39–42: o desafio (challenge) ocorre quando uma página protegida é solicitada sem um token válido. Por padrão, a resposta é um erro 401 sem conteúdo: útil para um cliente [JavaScript], mas confuso para um usuário. O usuário é redirecionado para a página de login. [HandleResponse()] informa a [JwtBearer] que a resposta está sendo tratada;
    • linhas 56–57: os dois componentes de middleware, nesta ordem: primeiro determinar quem é o usuário, depois verificar suas permissões.

    using Example25.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example25.Controllers;

public class AuthController(AuthService auth, IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => View("Login");

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password)
    {
        var user = auth.Validate(login, password);
        if (user is null)
        {
            // uma mensagem deliberadamente vaga: não dizemos se é o login ou a senha
            ViewData["Error"] = "Incorrect login or password.";
            ViewData["Login"] = login;
            return View("Login");
        }
        // o token, em um cookie que o JavaScript da página não consegue ler
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    // sair altera o estado do usuário: é um POST
    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }
}

    Vamos comentar este código:

    • linhas 14–23: o login. Se o login falhar, o formulário é exibido novamente com uma mensagem deliberadamente vaga: dizer “este login não existe” ajudaria um atacante a encontrar logins válidos;
    • linhas 26–32: em caso de sucesso, o token é armazenado em um cookie [HttpOnly]; o [JavaScript] da página não consegue lê-lo, o que protege o token caso um script malicioso seja executado na página. [SameSite = Lax]: o cookie não é incluído em requisições POST enviadas de outro site, o que protege contra a falsificação de requisições ([CSRF]). A duração do cookie corresponde à do token;
    • Linhas 36–40: o logout exclui o cookie ([Cookies.Delete]). Trata-se de um POST: ele altera o estado do usuário (veja o Exemplo 21).

    using System.Collections.Concurrent;

namespace Example27.Auth;

// um captcha: seu identificador e sua imagem (SVG)
public record Captcha(string Id, string Svg);

// singleton: os captchas pendentes são mantidos em memória pelo servidor
public class CaptchaService
{
    private static readonly TimeSpan Lifetime = TimeSpan.FromMinutes(10);

    // identificador -> (resposta esperada, data de expiração).
    // ConcurrentDictionary: um dicionário utilizável por várias requisições simultâneas
    private readonly ConcurrentDictionary<string, (string Text, DateTime Expires)> pending = new();

    public Captcha Generate()
    {
        var text = CaptchaDrawing.RandomText(5);
        var id = Guid.NewGuid().ToString(); // um identificador aleatório, impossível de adivinhar
        pending[id] = (text, DateTime.UtcNow + Lifetime);
        return new Captcha(id, CaptchaDrawing.Svg(text));
    }

    // o captcha é CONSUMIDO em todos os casos: uma imagem é usada uma só vez
    public bool Check(string? id, string? input)
    {
        if (id is null || input is null || !pending.TryRemove(id, out var expected))
        {
            return false;
        }
        return expected.Expires > DateTime.UtcNow
            && string.Equals(expected.Text, input.Trim(), StringComparison.OrdinalIgnoreCase);
    }
}

    Vamos detalhar este código:

    • linha 15: as respostas esperadas são armazenadas pelo servidor sob um identificador aleatório. A página contém apenas a imagem e o identificador: a resposta nunca sai do servidor. O serviço é um singleton, usado por requisições concorrentes: [ConcurrentDictionary] é um dicionário projetado para essa finalidade;
    • linhas 17–22: uma cadeia de cinco caracteres é sorteada aleatoriamente, armazenada e desenhada como imagem. O desenho é feito pela classe [CaptchaDrawing] (arquivo [Auth/CaptchaDrawing.cs], extraída do estudo de caso e comentada linha por linha): cada caractere é um conjunto de linhas que é ampliado, inclinado e distorcido aleatoriamente, e a imagem é borrada por três curvas. Caracteres facilmente confundidos (0 e O, 1, I e L) são excluídos;
    • linhas 26–33: o processo de verificação consome o CAPTCHA ([TryRemove] o remove do dicionário): uma imagem é usada uma única vez; um bot não pode reutilizar uma solução que já encontrou. Letras maiúsculas não contam.

    namespace Example27.Auth;

// singleton: conta as falhas de login de cada endereço IP
public class LimiterService
{
    private const int MaxFailures = 3;
    private static readonly TimeSpan BlockDuration = TimeSpan.FromMinutes(1);

    private class Counter
    {
        public int Failures;
        public DateTime BlockedUntil;
    }

    private readonly Dictionary<string, Counter> counters = [];
    private readonly Lock padlock = new(); // várias requisições simultâneas: um acesso por vez

    // o número de segundos de bloqueio restantes (0: não bloqueado)
    public int BlockedSeconds(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c)) return 0;
            var remaining = c.BlockedUntil - DateTime.UtcNow;
            return remaining > TimeSpan.Zero ? (int)Math.Ceiling(remaining.TotalSeconds) : 0;
        }
    }

    public void Failure(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c))
            {
                c = new Counter();
                counters[ip] = c;
            }
            c.Failures++;
            if (c.Failures >= MaxFailures)
            {
                c.BlockedUntil = DateTime.UtcNow + BlockDuration;
                c.Failures = 0; // após o bloqueio, recomeçamos do zero
            }
        }
    }

    public void Success(string ip)
    {
        lock (padlock)
        {
            counters.Remove(ip);
        }
    }
}
    • linhas 6–7: três tentativas malsucedidas bloqueiam o endereço IP por um minuto;
    • linha 16: o dicionário de contadores é compartilhado por todas as requisições. A instrução [lock (padlock) { ... }] garante que apenas uma requisição por vez execute o bloco: duas falhas simultâneas não podem se perder (veja [Interlocked], Capítulo 6). [Lock] é o tipo de lock para o [.NET] 9 e versões posteriores;
    • linhas 19–25: o tempo de bloqueio restante;
    • linhas 29–42: uma falha é contabilizada; na terceira falha, o endereço é bloqueado;
    • linhas 47–51: um sucesso zera o contador.

    using Example27.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example27.Controllers;

public class AuthController(AuthService auth, CaptchaService captchas, LimiterService limiter,
    IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => LoginPage(null, null);

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password, string? captchaId, string? captcha)
    {
        // o endereço IP do cliente
        var ip = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "?";

        // 1. um endereço bloqueado nem sequer é atendido
        var seconds = limiter.BlockedSeconds(ip);
        if (seconds > 0)
        {
            return LoginPage(login, $"Too many attempts: try again in {seconds} s.");
        }
        // 2. primeiro o captcha (barato), depois a senha (bcrypt, caro)
        if (!captchas.Check(captchaId, captcha))
        {
            limiter.Failure(ip);
            return LoginPage(login, "The image code is incorrect.");
        }
        var user = auth.Validate(login, password);
        if (user is null)
        {
            limiter.Failure(ip);
            return LoginPage(login, "Incorrect login or password.");
        }
        limiter.Success(ip);
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }

    // cada exibição do formulário constrói um NOVO captcha
    private ViewResult LoginPage(string? login, string? error)
    {
        ViewData["Login"] = login;
        ViewData["Error"] = error;
        return View("Login", captchas.Generate());
    }
}

    Vamos comentar este código:

    • linha 19: o endereço do cliente, IP, fornecido pela conexão de rede;
    • linhas 22–25: um endereço bloqueado nem sequer é atendido;
    • linhas 28–31: o CAPTCHA é verificado primeiro e, em seguida, a senha (uma operação [bcrypt] computacionalmente cara) somente se o CAPTCHA estiver correto. Um CAPTCHA incorreto conta como uma falha;
    • linhas 57–61: a cada exibição do formulário, um novo CAPTCHA é gerado, que se torna o modelo da view.
